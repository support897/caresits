// - a signed-in user touches only their own users/{uid} folder

// - _activity/{uid} is the photo lifecycle stamp (Ilse's ruling

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isOwner() {
      return signedIn()
        && exists(/databases/$(database)/documents/_owner/config)
        && request.auth.uid in
           get(/databases/$(database)/documents/_owner/config).data.uids;
    }
    function isAnon() {
      return signedIn()
        && request.auth.token.firebase.sign_in_provider == 'anonymous';
    }
    function demoUnlocked() {
      return exists(/databases/$(database)/documents/_unlocks/$(request.auth.uid))
        && exists(/databases/$(database)/documents/_demo/config)
        && get(/databases/$(database)/documents/_unlocks/$(request.auth.uid)).data.codeHash
           == get(/databases/$(database)/documents/_demo/config).data.codeHash;
    }


    function aggPairedSignup() {
      return signedIn() && !isAnon()
        && !exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && existsAfter(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function aggPairedActivation() {
      return signedIn() && !isAnon()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && !('activatedAt' in
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data)
        && 'activatedAt' in
           getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function aggPairedReturn() {
      return signedIn() && !isAnon()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid))
             .data.get('return7dCounted', false) == false
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid))
             .data.get('return7dCounted', false) == true;
    }
    function aggPlusOne(field) {
      // F17 Block B: a transform's value is unreadable in
      // request.resource; the counter doc's own getAfter state
      // is where transforms resolve, so the +1 is judged there.
      return !(field in
               request.resource.data.diff(resource.data).affectedKeys())
        || getAfter(/databases/$(database)/documents/_owner/aggregates/people/main)
             .data.get(field, 0) == resource.data.get(field, 0) + 1;
    }

    // delta). The grant targets are enumerated as three explicit
    // matches below — identical semantics to the delta's tail
    // pattern, with no new wildcards.
    function relId(a, b) { return a < b ? a + "_" + b : b + "_" + a; }
    function relData(a, b) {
      return get(/databases/$(database)/documents/relationships/$(relId(a, b))).data;
    }
    function relActive(a, b) {
      return exists(/databases/$(database)/documents/relationships/$(relId(a, b)))
        && relData(a, b).state in ["contact", "hired"]
        && !blockedEitherWay(a, b);
    }


    // Calendar helpers (Amendment 5 delta, cleared v3).
    // blockedEitherWay is realised here from Security's safety
    // ruling ahead of the safety delta's own pass; its text is
    // flagged for that read.
    function blockedEitherWay(a, b) {
      return exists(/databases/$(database)/documents/users/$(a)/blocks/$(b))
          || exists(/databases/$(database)/documents/users/$(b)/blocks/$(a));
    }
    function pairDoc(pairId) {
      return get(/databases/$(database)/documents/relationships/$(pairId));
    }
    function pairRel(pairId) {
      return exists(/databases/$(database)/documents/relationships/$(pairId))
        ? pairDoc(pairId).data : null;
    }
    function pairOpen(pairId) {
      return signedIn() && !isAnon()
        && pairRel(pairId) != null
        && pairRel(pairId).state in ["contact", "hired"]
        && request.auth.uid in pairRel(pairId).members
        && !blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid);
    }
    function pairMember(pairId) {
      return signedIn() && pairRel(pairId) != null
        && request.auth.uid in pairRel(pairId).members;
    }
    function pairSevered(pairId) {
      return pairRel(pairId) != null
        && (pairRel(pairId).state == "ended"
            || blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid));
    }
    function bookingRelPath(pairId) {
      return /databases/$(database)/documents/relationships/$(pairId);
    }
    function bookingRelData(pairId) {
      return exists(bookingRelPath(pairId))
        ? get(bookingRelPath(pairId)).data
        : getAfter(bookingRelPath(pairId)).data;
    }
    function pendingBookingPairMatches(pairId) {
      let pair = getAfter(bookingRelPath(pairId)).data;
      return pair.state == "pending"
        && pair.createdBy == request.auth.uid
        && pair.familyUid == request.auth.uid
        && request.auth.uid in pair.members
        && !blockedEitherWay(pair.familyUid, pair.carerUid);
    }
    function newPendingBookingPair(pairId) {
      return signedIn() && !isAnon()
        && !exists(bookingRelPath(pairId))
        && existsAfter(bookingRelPath(pairId))
        && pendingBookingPairMatches(pairId);
    }
    function bookingPairCanCreate(pairId) {
      return pairOpen(pairId) || newPendingBookingPair(pairId);
    }
    function bookingPairCanRead(pairId) {
      return pairOpen(pairId)
        || (signedIn() && !isAnon() && pairMember(pairId) && !pairSevered(pairId));
    }

function validAuPairHost(host) {
  return (host is map)
    ? (host.keys().hasOnly(["country", "room", "pocketMoneyPerWeek", "hoursPerWeek", "childcareDuties", "otherHousework", "languages", "visaHelp", "infantCareNeeded"])
      ? ((!("country" in host) || (host.country is string && host.country.size() <= 80))
        && (!("room" in host) || (host.room is string && host.room.size() <= 300))
        && (!("pocketMoneyPerWeek" in host) || host.pocketMoneyPerWeek == null || (host.pocketMoneyPerWeek is number && host.pocketMoneyPerWeek >= 0))
        && (!("hoursPerWeek" in host) || host.hoursPerWeek == null || (host.hoursPerWeek is number && host.hoursPerWeek >= 0))
        && (!("childcareDuties" in host) || (host.childcareDuties is list && host.childcareDuties.size() <= 30))
        && (!("otherHousework" in host) || (host.otherHousework is list && host.otherHousework.size() <= 30))
        && (!("languages" in host) || (host.languages is list && host.languages.size() <= 20))
        && (!("visaHelp" in host) || host.visaHelp is bool)
        && (!("infantCareNeeded" in host) || host.infantCareNeeded is bool))
      : false)
    : false;
}
function validAlsoSide(other, primarySide) {
  return (other is map)
    ? (other.keys().hasOnly(["side", "about", "headline", "photoUrl", "videoKey", "experienceYears", "rate", "availability", "careTypes", "kids", "languages", "suburb", "scheduleNeed", "budgetMax", "maxDistanceKm", "style", "activities"])
      ? (other.side in ["family", "carer"]
        && other.side != primarySide
        && (!("about" in other) || (other.about is string && other.about.size() <= 300))
        && (!("headline" in other) || (other.headline is string && other.headline.size() <= 80))
        && (!("photoUrl" in other) || (other.photoUrl is string && other.photoUrl.size() <= 2000))
        && (!("videoKey" in other) || (other.videoKey is string && other.videoKey.size() <= 500))
        && (!("experienceYears" in other) || (other.experienceYears is number && other.experienceYears >= 0))
        && (!("rate" in other) || (other.rate is number && other.rate >= 0))
        && (!("careTypes" in other) || (other.careTypes is list && other.careTypes.size() <= 3 && other.careTypes.hasOnly(["babysitter", "nanny", "au_pair"])))
        && (!("kids" in other) || (other.kids is list && other.kids.size() <= 8))
        && (!("suburb" in other) || (other.suburb is string && other.suburb.size() <= 80))
        && (!("budgetMax" in other) || (other.budgetMax is number && other.budgetMax >= 0))
        && (!("maxDistanceKm" in other) || (other.maxDistanceKm is number && other.maxDistanceKm >= 0)))
      : false)
    : false;
}

match /schedule/{pairId} {
  allow read, write: if false;  // the pair doc itself carries nothing

  // BLOCK VOCABULARY (a fence, read as one):
  //   kind:  "extra"    a one time added session
  //          "off"      a skipped session or day marker
  //          "moved"    a session held at a new time, one time
  //   state: "live" — blocks are never edited and never deleted.
  // "day" is an ISO date (YYYY-MM-DD). The regular week's shape
  // does NOT live here; it lives in the week versions below.
  match /blocks/{blockId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["day", "start", "end", "kind", "withUid", "state", "sourceRequest", "createdBy", "createdAt"])
      && request.resource.data.kind in ["extra", "off", "moved"]
      && request.resource.data.state == "live"
      && request.resource.data.createdBy == request.auth.uid
      && request.resource.data.withUid in pairDoc(pairId).data.members
      && request.resource.data.withUid != request.auth.uid
      // An acceptance landing: the named request stands
      // accepted, was decided by THIS writer, and the block
      // EQUALS the proposal: the kind maps (extra to extra,
      // off to off, move to moved), the day is the request's
      // day or one of its listed days, and start and end match.
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.state == "accepted"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.decidedBy == request.auth.uid
      && ((get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "extra" && request.resource.data.kind == "extra")
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "off" && request.resource.data.kind == "off")
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "move" && request.resource.data.kind == "moved"))
      && (request.resource.data.day == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.day
          || ("days" in get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data
              && request.resource.data.day in get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.days))
      && request.resource.data.get("start", "") == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.get("start", "")
      && request.resource.data.get("end", "") == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.get("end", "");
    allow update: if false;
    allow delete: if false;
  }

  // Requests: proposals only. A request doc can change its own
  // state (and its decider stamp) and nothing else, ever.
  // Kinds: "extra" · "off" · "move" · "week-change".
  // A "week-change" request also carries proposedVersionId and
  // effectiveFrom; no other kind may carry them.
  match /requests/{reqId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["kind", "askedBy", "day", "days", "start", "end", "state", "decidedBy",
         "proposedVersionId", "effectiveFrom", "closeReason", "closedBy", "closedAt", "createdAt"])
      && request.resource.data.kind in ["extra", "off", "move", "week-change"]
      && request.resource.data.askedBy == request.auth.uid
      && request.resource.data.state == "waiting"
      && !("decidedBy" in request.resource.data)
      && !("closeReason" in request.resource.data)
      && !("closedBy" in request.resource.data)
      && !("closedAt" in request.resource.data)
      && (("proposedVersionId" in request.resource.data)
            == (request.resource.data.kind == "week-change"))
      && (("effectiveFrom" in request.resource.data)
            == (request.resource.data.kind == "week-change"));
    allow update: if
      // The living pair: decide or withdraw. The two arms
      // carry different fences: a withdrawal may not stamp
      // decidedBy; only a decision carries it.
      (pairOpen(pairId)
        && resource.data.state == "waiting"
        && (
          (request.resource.data.state in ["accepted", "declined"]
            && request.auth.uid != resource.data.askedBy
            && request.resource.data.decidedBy == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasAll(["state", "decidedBy"])
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["state", "decidedBy"]))
          || (request.resource.data.state == "withdrawn"
            && request.auth.uid == resource.data.askedBy
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["state"]))))
      ||
      // PIN A: the severed pair's records act. From "waiting"
      // only, exactly once, fenced to the stamp fields.
      (pairMember(pairId) && pairSevered(pairId)
        && resource.data.state == "waiting"
        && request.resource.data.state == "closed"
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(["state", "closeReason", "closedBy", "closedAt"])
        && request.resource.data.closedBy == request.auth.uid
        && ((request.resource.data.closeReason == "ended"
             && pairRel(pairId).state == "ended")
            || (request.resource.data.closeReason == "blocked"
             && blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid))));
    allow delete: if false;
  }

  // Week versions: write once proposals. The pattern is child
  // bare: a list of entries, each only a weekday code, a day
  // part, a start and an end. Never updated, never deleted.
  match /weekVersions/{versionId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["pattern", "effectiveFrom", "proposedBy", "sourceRequest", "createdAt"])
      && request.resource.data.proposedBy == request.auth.uid
      && request.resource.data.pattern is list
      && (!("sourceRequest" in request.resource.data)
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "week-change"
              && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.askedBy == request.auth.uid));
    allow update: if false;
    allow delete: if false;
  }

  // The week pointer: which version is the agreed week, and
  // from when. Created at formation; after that it moves only
  // on the accepted week-change binding.
  match /weekCurrent/{doc} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId) && doc == "main"
      && request.resource.data.keys().hasOnly(
        ["versionId", "effectiveFrom", "updatedBy", "updatedAt"])
      && request.resource.data.updatedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/weekVersions/$(request.resource.data.versionId)).data.proposedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/weekVersions/$(request.resource.data.versionId)).data.effectiveFrom == request.resource.data.effectiveFrom;
    // Every later move rides one accepted week-change request,
    // named on the pointer itself in boundRequest, and the
    // pointer's new values must EQUAL that request's proposal.
    allow update: if pairOpen(pairId) && doc == "main"
      && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(["versionId", "effectiveFrom", "updatedBy", "updatedAt", "boundRequest"])
      && request.resource.data.updatedBy == request.auth.uid
      && ("boundRequest" in request.resource.data)
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.kind == "week-change"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.state == "accepted"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.decidedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.askedBy != request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.proposedVersionId == request.resource.data.versionId
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.effectiveFrom == request.resource.data.effectiveFrom;
    allow delete: if false;
  }
}

          && csRole(spaceId) in ["admin", "regular", "limited"]
          && request.resource.data.keys().hasOnly(["childId", "day", "kind", "at", "text", "mood", "photoKey", "by", "asks"])
          && request.resource.data.by == request.auth.uid
          && (!(request.resource.data.kind == "photo" || "photoKey" in request.resource.data)
              || csAdmin(spaceId) || csPerm(spaceId, "photos"));
        allow update: if (csMember(spaceId)
            && csRole(spaceId) in ["admin", "regular", "limited"]
            && request.resource.data.keys().hasOnly(["childId", "day", "kind", "at", "text", "mood", "photoKey", "by", "asks"])
            && request.resource.data.by == request.auth.uid
            && (!(request.resource.data.kind == "photo" || "photoKey" in request.resource.data)
                || csAdmin(spaceId) || csPerm(spaceId, "photos")))
          || (csMember(spaceId)
            && csRole(spaceId) in ["admin", "regular", "limited"]
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["asks"])
            && request.resource.data.asks is list
            && request.resource.data.asks.size() <= 20
            && (csAdmin(spaceId) || csPerm(spaceId, "photos")));
        allow delete: if csAdmin(spaceId);
      }
      match /calendar/{id} {
        allow read: if csMember(spaceId) && (csAdmin(spaceId) || csPerm(spaceId, "calendar"));
        allow create: if csMember(spaceId)
          && csRole(spaceId) in ["admin", "regular", "limited"]
          && request.resource.data.keys().hasOnly(["kind", "title", "start", "end", "childId", "ownerUid", "state", "swap", "bookingRef"])
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.kind in ["shift", "school", "activity", "pickup", "timeoff"]
          && ((request.resource.data.kind == "timeoff" && request.resource.data.state == "requested")
              || (request.resource.data.kind != "timeoff" && request.resource.data.state in ["active", "requested"]));
        allow update: if (csAdmin(spaceId)
            && resource.data.kind == "timeoff"
            && resource.data.state == "requested"
            && request.resource.data.kind == "timeoff"
            && request.resource.data.state in ["approved", "declined", "swap"]
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["state", "swap"]))
          || (csAdmin(spaceId)
            && resource.data.kind != "timeoff" && resource.data.kind != "booking"
            && request.resource.data.kind == resource.data.kind
            && request.resource.data.state == resource.data.state
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(
              ["title", "start", "end", "childId", "ownerUid", "swap", "bookingRef"]))
          || (csMember(spaceId) && csRole(spaceId) in ["regular", "limited"]
            && resource.data.kind != "timeoff" && resource.data.kind != "booking"
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.kind == resource.data.kind
            && request.resource.data.state == resource.data.state
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["title", "start", "end", "bookingRef"]));
        allow delete: if false;
      }
      match /shopping/{id} {
        allow read: if csMember(spaceId);
        allow create: if csMember(spaceId) && csRole(spaceId) in ["admin", "regular"]
          && request.resource.data.keys().hasOnly(["item", "childId", "source", "sourceId", "buyer", "due", "bought", "reimbursed"])
          && request.resource.data.keys().hasAll(["item", "source", "bought", "reimbursed"])
          && request.resource.data.item is string && request.resource.data.item.size() > 0 && request.resource.data.item.size() <= 200
          && request.resource.data.source in ["meal", "activity", "manual"]
          && request.resource.data.bought is bool && request.resource.data.reimbursed is bool
          && (!("childId" in request.resource.data) || request.resource.data.childId is string)
          && (!("sourceId" in request.resource.data) || request.resource.data.sourceId is string)
          && (!("buyer" in request.resource.data) || request.resource.data.buyer is string)
          && (!("due" in request.resource.data) || request.resource.data.due is string);
        allow update: if csMember(spaceId) && csRole(spaceId) in ["admin", "regular"]
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["buyer", "due", "bought", "reimbursed"])
          && (!("buyer" in request.resource.data) || request.resource.data.buyer is string)
          && (!("due" in request.resource.data) || request.resource.data.due is string)
          && (!("bought" in request.resource.data) || request.resource.data.bought is bool)
          && (!("reimbursed" in request.resource.data) || request.resource.data.reimbursed is bool);
        allow delete: if csAdmin(spaceId);
      }
      match /mealPlan/{day} {
        allow read: if csMember(spaceId);
        allow create, update: if csAdmin(spaceId)
          && request.resource.data.keys().hasOnly(["meals"])
          && request.resource.data.meals is list
          && request.resource.data.meals.size() <= 50;
        allow delete: if csAdmin(spaceId);
      }
      match /checkins/{checkinId} {
        allow read: if csMember(spaceId);
        allow create: if csMember(spaceId)
          && csRole(spaceId) in ["regular", "limited"]
          && request.resource.data.keys().hasOnly(["by", "kind", "bookingId", "at"])
          && request.resource.data.keys().hasAll(["by", "kind", "bookingId", "at"])
          && request.resource.data.by == request.auth.uid
          && request.resource.data.kind in ["arrived", "leaving", "help"]
          && request.resource.data.at == request.time
          && request.resource.data.bookingId is string
          && request.resource.data.bookingId.size() <= 128
          && directSpaceCheckinAllowed(spaceId, request.resource.data.bookingId);
        allow update, delete: if false;
      }
      match /messages/{id} {
        allow read: if csMember(spaceId) && csRole(spaceId) != "viewer"
          && (csAdmin(spaceId) || csPerm(spaceId, "privateMessages"));
        allow create: if csMember(spaceId) && csRole(spaceId) != "viewer"
          && csRole(spaceId) in ["admin", "regular", "limited"]
          && (csAdmin(spaceId) || csPerm(spaceId, "privateMessages"))
          && request.resource.data.keys().hasOnly(["from", "text", "at", "kind", "img"])
          && request.resource.data.from == request.auth.uid
          && request.resource.data.text is string && request.resource.data.text.size() <= 1000
          && request.resource.data.at == request.time
          && request.resource.data.kind in ["text", "photo", "photoAsk"]
          && (!("img" in request.resource.data)
              || (request.resource.data.img is string && request.resource.data.img.size() <= 400000));
        allow update, delete: if false;
      }
      match /medical/{childId} {
        allow read: if csAdmin(spaceId) || (csPerm(spaceId, "allergies") && csPerm(spaceId, "medication"));
        allow create, update, delete: if false;
        match /allergies/{recordId} {
          allow read: if csAdmin(spaceId) || csPerm(spaceId, "allergies");
          allow create, update: if csAdmin(spaceId)
            && request.resource.data.keys().hasOnly(["items", "updatedBy", "updatedAt", "alertId"])
            && request.resource.data.items is list && request.resource.data.items.size() <= 50
            && medicalRecordAlertMatches(spaceId, childId, "allergies", recordId);
          allow delete: if false;
        }
        match /medication/{recordId} {
          allow read: if csAdmin(spaceId) || csPerm(spaceId, "medication");
          allow create, update: if csAdmin(spaceId)
            && request.resource.data.keys().hasOnly(["items", "updatedBy", "updatedAt", "alertId"])
            && request.resource.data.items is list && request.resource.data.items.size() <= 50
            && medicalRecordAlertMatches(spaceId, childId, "medication", recordId);
          allow delete: if false;
        }
        match /emergency/{recordId} {
          allow read: if csAdmin(spaceId);
          allow create, update: if csAdmin(spaceId)
            && request.resource.data.keys().hasOnly(["contacts", "doctor", "notes", "updatedBy", "updatedAt", "alertId"])
            && medicalRecordAlertMatches(spaceId, childId, "emergency", recordId);
          allow delete: if false;
        }
      }
      match /alerts/{alertId} {
        allow read: if csMember(spaceId);
        allow create: if csAdmin(spaceId)
          && request.resource.data.keys().hasOnly(["kind", "by", "at", "summary", "childId", "section", "recordId"])
          && request.resource.data.kind == "medical-change"
          && request.resource.data.by == request.auth.uid
          && request.resource.data.at == request.time
          && request.resource.data.summary is string && request.resource.data.summary.size() <= 240
          && existsAfter(/databases/$(database)/documents/careSpaces/$(spaceId)/medical/$(request.resource.data.childId)/$(request.resource.data.section)/$(request.resource.data.recordId))
          && getAfter(/databases/$(database)/documents/careSpaces/$(spaceId)/medical/$(request.resource.data.childId)/$(request.resource.data.section)/$(request.resource.data.recordId)).data.alertId == alertId
          && getAfter(/databases/$(database)/documents/careSpaces/$(spaceId)/medical/$(request.resource.data.childId)/$(request.resource.data.section)/$(request.resource.data.recordId)).data.updatedBy == request.auth.uid
          && getAfter(/databases/$(database)/documents/careSpaces/$(spaceId)/medical/$(request.resource.data.childId)/$(request.resource.data.section)/$(request.resource.data.recordId)).data.updatedAt == request.time;
        allow update, delete: if false;

    }
    match /careSpaceInvites/{inviteId} {
      allow get: if signedIn() && !isAnon();
      allow list: if false;
      allow create: if signedIn() && !isAnon()
        && request.resource.data.keys().hasOnly(["spaceId", "role", "createdBy", "createdAt", "expiresAt", "usedBy", "perms", "targetUid", "bookingId"])

        && request.resource.data.spaceId is string
        && request.resource.data.spaceId.matches("^[^/]{1,128}$")

        && resource.data.createdBy != request.auth.uid
        && (!("targetUid" in resource.data) || resource.data.targetUid == request.auth.uid)

        && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.role == resource.data.role
        && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.invitedBy == resource.data.createdBy
        && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.status == "active"
        && (("perms" in resource.data && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.perms == resource.data.perms)
            || (!("perms" in resource.data)
                && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.perms.keys().hasOnly(["allergies", "medication", "photos", "calendar", "privateMessages"])
                && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.perms.get("allergies", false) == false

                && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.perms.get("photos", false) == false

                && getAfter(/databases/$(database)/documents/careSpaces/$(resource.data.spaceId)/members/$(request.auth.uid)).data.perms.get("privateMessages", false) == false));
