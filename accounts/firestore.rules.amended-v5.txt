// Canonical accounts-phase Security Rules v2.2 (identical for
// every app project + test twin). The rule IS the wall:
// - a signed-in user touches only their own users/{uid} folder
// - NO client ever writes a plan-shaped key (plan, planSource,
//   comp, role, owner): the user record carries no raisable
//   plan state; entitlement resolves from _comp (TW-3)
// - _owner/config + _comp/{uid} + _demo/config are desk/console
//   records: clients can neither read nor write them; the rules
//   read them via get() (owner-of-record + comp + demo hash)
// - _entitlements/{uid} is the payments chapter record (Finance
//   §3): written ONLY by the payment path on the admin plane. A
//   client may read its OWN record and can never write it. It sits
//   at the root on the _comp precedent, outside the users/{uid}
//   catch-all, whose plan-key ban could not fence its field names.
// - _activity/{uid} is the photo lifecycle stamp (Ilse's ruling
//   + HQ spec, 19 Oct 2026): the carer's lastActiveAt, refreshed
//   when the signed-in app opens. It sits at the root for the
//   same reason (the catch-all would let any value land), and
//   its write arm admits ONE key whose value must equal the
//   server's own clock, so a stamp can never be forged or
//   backdated. Clients never read it; the sweeper reads it on
//   the admin plane. Amendment v5 adds this arm alone to the
//   sealed v4 text; its case is RULES-AMENDMENT-LIFECYCLE.md.
// - the OWNER HAS NO READ PATH INTO USER FOLDERS (HQ ruling
//   15 Oct, the Fit shape): dashboards read PRECOMPUTED
//   AGGREGATES at _owner/aggregates only — numbers, never
//   people (TW-4 / NS-1). Aggregates writers land with the
//   dashboard builds; the surface never counts live records.
// - demo sessions (SPARK): an anonymous session may store
//   data ONLY after a valid unlock — the unlock record at
//   _unlocks/{uid} can be created solely by presenting the
//   access code's SHA-256, checked here via get() against
//   _demo/config (DS-2, rules-side). Connection paths, when
//   they land (P4), deny the anonymous provider claim by the
//   same pattern (DS-4, rules-side).
// - the write verbs are split (Security FAIL 1, 15 Oct):
//   create/update keep the plan-key condition; delete takes
//   the path-owner condition alone (request.resource is
//   absent on deletes).
// - everything else is denied by default (no match = no access)
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function isOwner() {
      return signedIn()
        && exists(/databases/$(database)/documents/_owner/config)
        && request.auth.uid in
           get(/databases/$(database)/documents/_owner/config).data.uids;
    }
    function isAnon() {
      return signedIn()
        && request.auth.token.firebase.sign_in_provider == 'anonymous';
    }
    function demoUnlocked() {
      return exists(/databases/$(database)/documents/_unlocks/$(request.auth.uid))
        && exists(/databases/$(database)/documents/_demo/config)
        && get(/databases/$(database)/documents/_unlocks/$(request.auth.uid)).data.codeHash
           == get(/databases/$(database)/documents/_demo/config).data.codeHash;
    }


    function aggPairedSignup() {
      return signedIn() && !isAnon()
        && !exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && existsAfter(/databases/$(database)/documents/users/$(request.auth.uid));
    }
    function aggPairedActivation() {
      return signedIn() && !isAnon()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && !('activatedAt' in
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data)
        && 'activatedAt' in
           getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    function aggPairedReturn() {
      return signedIn() && !isAnon()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid))
             .data.get('return7dCounted', false) == false
        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid))
             .data.get('return7dCounted', false) == true;
    }
    function aggPlusOne(field) {
      // F17 Block B: a transform's value is unreadable in
      // request.resource; the counter doc's own getAfter state
      // is where transforms resolve, so the +1 is judged there.
      return !(field in
               request.resource.data.diff(resource.data).affectedKeys())
        || getAfter(/databases/$(database)/documents/_owner/aggregates/people/main)
             .data.get(field, 0) == resource.data.get(field, 0) + 1;
    }


    // Matching chapter helpers (fold-in of the cleared matching
    // delta). The grant targets are enumerated as three explicit
    // matches below — identical semantics to the delta's tail
    // pattern, with no new wildcards.
    function relId(a, b) { return a < b ? a + "_" + b : b + "_" + a; }
    function relData(a, b) {
      return get(/databases/$(database)/documents/relationships/$(relId(a, b))).data;
    }
    function relActive(a, b) {
      return exists(/databases/$(database)/documents/relationships/$(relId(a, b)))
        && relData(a, b).state in ["contact", "hired"]
        && !blockedEitherWay(a, b);
    }


    // Calendar helpers (Amendment 5 delta, cleared v3).
    // blockedEitherWay is realised here from Security's safety
    // ruling ahead of the safety delta's own pass; its text is
    // flagged for that read.
    function blockedEitherWay(a, b) {
      return exists(/databases/$(database)/documents/users/$(a)/blocks/$(b))
          || exists(/databases/$(database)/documents/users/$(b)/blocks/$(a));
    }
    function pairDoc(pairId) {
      return get(/databases/$(database)/documents/relationships/$(pairId));
    }
    function pairRel(pairId) {
      return exists(/databases/$(database)/documents/relationships/$(pairId))
        ? pairDoc(pairId).data : null;
    }
    function pairOpen(pairId) {
      return signedIn() && !isAnon()
        && pairRel(pairId) != null
        && pairRel(pairId).state in ["contact", "hired"]
        && request.auth.uid in pairRel(pairId).members
        && !blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid);
    }
    function pairMember(pairId) {
      return signedIn() && pairRel(pairId) != null
        && request.auth.uid in pairRel(pairId).members;
    }
    function pairSevered(pairId) {
      return pairRel(pairId) != null
        && (pairRel(pairId).state == "ended"
            || blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid));
    }

    match /users/{uid}/{document=**} {
      allow read: if signedIn() && request.auth.uid == uid
        && (!isAnon() || demoUnlocked());
      allow create, update: if signedIn() && request.auth.uid == uid
        && (!isAnon() || demoUnlocked())
        && !request.resource.data.keys()
             .hasAny(['plan', 'planSource', 'comp', 'role', 'owner']);
      allow delete: if signedIn() && request.auth.uid == uid
        && (!isAnon() || demoUnlocked());
    }


    // Relationship grants: enumerated, direction checked. These
    // matches only ever ADD a read for the relationship partner;
    // writes stay with the wall above.
    match /users/{uid}/children/{childId} {
      allow read: if signedIn() && !isAnon()
        && relActive(request.auth.uid, uid)
        && relData(request.auth.uid, uid).carerUid == request.auth.uid
        && relData(request.auth.uid, uid).familyUid == uid;
    }
    match /users/{uid}/dayStory/{entryId} {
      allow read: if signedIn() && !isAnon()
        && relActive(request.auth.uid, uid)
        && relData(request.auth.uid, uid).carerUid == request.auth.uid
        && relData(request.auth.uid, uid).familyUid == uid
        && resource.data.day >= relData(request.auth.uid, uid).createdDay;
    }
    match /users/{uid}/documents/{docId} {
      allow read: if signedIn() && !isAnon()
        && relActive(request.auth.uid, uid)
        && relData(request.auth.uid, uid).familyUid == request.auth.uid
        && relData(request.auth.uid, uid).carerUid == uid;
    }

    // Fold (17 Oct, twin): the Recently Deleted bin, named in its own right.
    // Source: firestore.rules.recently-deleted.foldin.txt (owner-only; adds no new reach).
    match /users/{uid}/recentlyDeleted/{binId} {
      allow read: if signedIn() && request.auth.uid == uid && (!isAnon() || demoUnlocked());
      allow create, update: if signedIn() && request.auth.uid == uid && (!isAnon() || demoUnlocked()) && !request.resource.data.keys().hasAny(['plan', 'planSource', 'comp', 'role', 'owner']);
      allow delete: if signedIn() && request.auth.uid == uid && (!isAnon() || demoUnlocked());
    }


    // The demo unlock: creatable ONLY with the code's hash,
    // checked against the desk-held record; never editable.
    match /_unlocks/{uid} {
      allow create: if signedIn() && request.auth.uid == uid && isAnon()
        && exists(/databases/$(database)/documents/_demo/config)
        && request.resource.data.codeHash ==
           get(/databases/$(database)/documents/_demo/config).data.codeHash;
      allow read: if signedIn() && request.auth.uid == uid;
      allow update, delete: if false;
    }


    // CEO people counters: paired writes only (see header).
    // F17 Block B v3 (Security re-arm spec, 2026-10-19): the
    // signup counts (total + weekly) move to one-field docs
    // below (peopleTotal, peopleWeekly); this doc keeps only
    // activatedWithin7d + returning7d under their existing
    // arms (sole-transform writes, proven). Its total and
    // weekly fields are retired: no client writes them. Every
    // other aggregates document stays client-unwritable: money is
    // written by the payment path alone, the rest by desk feeds.
    match /_owner/aggregates/people/main {
      allow read: if isOwner();
      allow create: if
        (aggPairedActivation()
          && request.resource.data.keys()
               .hasOnly(['activatedWithin7d'])
          && getAfter(/databases/$(database)/documents/_owner/aggregates/people/main)
               .data.get('activatedWithin7d', 0) == 1)
        || (aggPairedReturn()
          && request.resource.data.keys()
               .hasOnly(['returning7d'])
          && getAfter(/databases/$(database)/documents/_owner/aggregates/people/main)
               .data.get('returning7d', 0) == 1);
      allow update: if
        (aggPairedActivation()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['activatedWithin7d'])
          && aggPlusOne('activatedWithin7d'))
        || (aggPairedReturn()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(['returning7d'])
          && aggPlusOne('returning7d'));
      allow delete: if false;
    }


    match /_owner/aggregates/peopleTotal/main {
      allow read: if isOwner();
      allow create: if aggPairedSignup()
        && request.resource.data.keys().hasOnly(['total'])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/peopleTotal/main)
             .data.get('total', 0) == 1;
      allow update: if aggPairedSignup()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['total'])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/peopleTotal/main)
             .data.get('total', 0) == resource.data.get('total', 0) + 1;
      allow delete: if false;
    }
    match /_owner/aggregates/peopleWeekly/{wk} {
      allow read: if isOwner();
      allow create: if aggPairedSignup()
        && wk.matches('^\\d{4}-W\\d{2}$')
        && request.resource.data.keys().hasOnly(['count'])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/peopleWeekly/$(wk))
             .data.get('count', 0) == 1;
      allow update: if aggPairedSignup()
        && wk.matches('^\\d{4}-W\\d{2}$')
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['count'])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/peopleWeekly/$(wk))
             .data.get('count', 0) == resource.data.get('count', 0) + 1;
      allow delete: if false;
    }


    // The safety counts: numbers only, never content. Filed
    // rises with a filing; open rises with it and drains by
    // exactly one on the operator's closure move. (Fold-in note
    // for Security's item 5 read: the drain arm is owner-only
    // and exactly minus one; the same-batch pairing with the
    // disposition write is the app layer's obligation, verified
    // behaviourally in the safety battery, because a rules
    // expression cannot name the report being closed.)
    match /_owner/aggregates/safetyFiled/main {
      allow read: if isOwner();
      allow create: if signedIn() && !isAnon()
        && request.resource.data.keys().hasOnly(["filed"])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyFiled/main)
             .data.get("filed", 0) == 1;
      allow update: if signedIn() && !isAnon()
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(["filed"])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyFiled/main)
             .data.get("filed", 0) == resource.data.get("filed", 0) + 1;
      allow delete: if false;
    }
    match /_owner/aggregates/safetyOpen/main {
      allow read: if isOwner();
      allow create: if signedIn() && !isAnon()
        && request.resource.data.keys().hasOnly(["open"])
        && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyOpen/main)
             .data.get("open", 0) == 1;
      allow update: if
        (signedIn() && !isAnon()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(["open"])
          && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyOpen/main)
               .data.get("open", 0) == resource.data.get("open", 0) + 1)
        || (isOwner()
          && request.resource.data.diff(resource.data).affectedKeys()
               .hasOnly(["open"])
          && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyOpen/main)
               .data.get("open", 0) == resource.data.get("open", 0) - 1
          && getAfter(/databases/$(database)/documents/_owner/aggregates/safetyOpen/main)
               .data.get("open", 0) >= 0);
      allow delete: if false;
    }

    // Dashboards read aggregates here and nowhere else.
    match /_owner/aggregates/{document=**} {
      allow read: if isOwner();
      allow write: if false;
    }

match /schedule/{pairId} {
  allow read, write: if false;  // the pair doc itself carries nothing

  // BLOCK VOCABULARY (a fence, read as one):
  //   kind:  "extra"    a one time added session
  //          "off"      a skipped session or day marker
  //          "moved"    a session held at a new time, one time
  //   state: "live" — blocks are never edited and never deleted.
  // "day" is an ISO date (YYYY-MM-DD). The regular week's shape
  // does NOT live here; it lives in the week versions below.
  match /blocks/{blockId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["day", "start", "end", "kind", "withUid", "state", "sourceRequest", "createdBy", "createdAt"])
      && request.resource.data.kind in ["extra", "off", "moved"]
      && request.resource.data.state == "live"
      && request.resource.data.createdBy == request.auth.uid
      && request.resource.data.withUid in pairDoc(pairId).data.members
      && request.resource.data.withUid != request.auth.uid
      // An acceptance landing: the named request stands
      // accepted, was decided by THIS writer, and the block
      // EQUALS the proposal: the kind maps (extra to extra,
      // off to off, move to moved), the day is the request's
      // day or one of its listed days, and start and end match.
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.state == "accepted"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.decidedBy == request.auth.uid
      && ((get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "extra" && request.resource.data.kind == "extra")
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "off" && request.resource.data.kind == "off")
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "move" && request.resource.data.kind == "moved"))
      && (request.resource.data.day == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.day
          || ("days" in get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data
              && request.resource.data.day in get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.days))
      && request.resource.data.get("start", "") == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.get("start", "")
      && request.resource.data.get("end", "") == get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.get("end", "");
    allow update: if false;
    allow delete: if false;
  }

  // Requests: proposals only. A request doc can change its own
  // state (and its decider stamp) and nothing else, ever.
  // Kinds: "extra" · "off" · "move" · "week-change".
  // A "week-change" request also carries proposedVersionId and
  // effectiveFrom; no other kind may carry them.
  match /requests/{reqId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["kind", "askedBy", "day", "days", "start", "end", "state", "decidedBy",
         "proposedVersionId", "effectiveFrom", "closeReason", "closedBy", "closedAt", "createdAt"])
      && request.resource.data.kind in ["extra", "off", "move", "week-change"]
      && request.resource.data.askedBy == request.auth.uid
      && request.resource.data.state == "waiting"
      && !("decidedBy" in request.resource.data)
      && !("closeReason" in request.resource.data)
      && !("closedBy" in request.resource.data)
      && !("closedAt" in request.resource.data)
      && (("proposedVersionId" in request.resource.data)
            == (request.resource.data.kind == "week-change"))
      && (("effectiveFrom" in request.resource.data)
            == (request.resource.data.kind == "week-change"));
    allow update: if
      // The living pair: decide or withdraw. The two arms
      // carry different fences: a withdrawal may not stamp
      // decidedBy; only a decision carries it.
      (pairOpen(pairId)
        && resource.data.state == "waiting"
        && (
          (request.resource.data.state in ["accepted", "declined"]
            && request.auth.uid != resource.data.askedBy
            && request.resource.data.decidedBy == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasAll(["state", "decidedBy"])
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["state", "decidedBy"]))
          || (request.resource.data.state == "withdrawn"
            && request.auth.uid == resource.data.askedBy
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(["state"]))))
      ||
      // PIN A: the severed pair's records act. From "waiting"
      // only, exactly once, fenced to the stamp fields.
      (pairMember(pairId) && pairSevered(pairId)
        && resource.data.state == "waiting"
        && request.resource.data.state == "closed"
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(["state", "closeReason", "closedBy", "closedAt"])
        && request.resource.data.closedBy == request.auth.uid
        && ((request.resource.data.closeReason == "ended"
             && pairRel(pairId).state == "ended")
            || (request.resource.data.closeReason == "blocked"
             && blockedEitherWay(pairRel(pairId).familyUid, pairRel(pairId).carerUid))));
    allow delete: if false;
  }

  // Week versions: write once proposals. The pattern is child
  // bare: a list of entries, each only a weekday code, a day
  // part, a start and an end. Never updated, never deleted.
  match /weekVersions/{versionId} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId)
      && request.resource.data.keys().hasOnly(
        ["pattern", "effectiveFrom", "proposedBy", "sourceRequest", "createdAt"])
      && request.resource.data.proposedBy == request.auth.uid
      && request.resource.data.pattern is list
      && (!("sourceRequest" in request.resource.data)
          || (get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.kind == "week-change"
              && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.sourceRequest)).data.askedBy == request.auth.uid));
    allow update: if false;
    allow delete: if false;
  }

  // The week pointer: which version is the agreed week, and
  // from when. Created at formation; after that it moves only
  // on the accepted week-change binding.
  match /weekCurrent/{doc} {
    allow read: if pairOpen(pairId);
    allow create: if pairOpen(pairId) && doc == "main"
      && request.resource.data.keys().hasOnly(
        ["versionId", "effectiveFrom", "updatedBy", "updatedAt"])
      && request.resource.data.updatedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/weekVersions/$(request.resource.data.versionId)).data.proposedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/weekVersions/$(request.resource.data.versionId)).data.effectiveFrom == request.resource.data.effectiveFrom;
    // Every later move rides one accepted week-change request,
    // named on the pointer itself in boundRequest, and the
    // pointer's new values must EQUAL that request's proposal.
    allow update: if pairOpen(pairId) && doc == "main"
      && request.resource.data.diff(resource.data).affectedKeys()
           .hasOnly(["versionId", "effectiveFrom", "updatedBy", "updatedAt", "boundRequest"])
      && request.resource.data.updatedBy == request.auth.uid
      && ("boundRequest" in request.resource.data)
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.kind == "week-change"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.state == "accepted"
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.decidedBy == request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.askedBy != request.auth.uid
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.proposedVersionId == request.resource.data.versionId
      && get(/databases/$(database)/documents/schedule/$(pairId)/requests/$(request.resource.data.boundRequest)).data.effectiveFrom == request.resource.data.effectiveFrom;
    allow delete: if false;
  }
}

  allow create: if signedIn() && !isAnon()
    && request.resource.data.keys().hasOnly(
      ["reporterUid", "reportedUid", "surface", "category", "reporterWords",
       "snapshot", "state", "createdAt"])
    && request.resource.data.reporterUid == request.auth.uid
    && request.resource.data.reportedUid != request.auth.uid
    && request.resource.data.surface in ["message", "card", "job", "profile"]
    && request.resource.data.category in
      ["spam_scam", "rude_abusive", "fake_profile", "child_safety", "inappropriate", "other"]
    && request.resource.data.reporterWords is string
    && request.resource.data.reporterWords.size() <= 1000
    && request.resource.data.snapshot is map
    && request.resource.data.snapshot.keys().hasOnly(["items", "text", "title"])
    && request.resource.data.snapshot.get("items", []) is list
    && request.resource.data.snapshot.get("items", []).size() <= 10
    && request.resource.data.snapshot.get("text", "") is string
    && request.resource.data.snapshot.get("text", "").size() <= 4000
    && request.resource.data.snapshot.get("title", "") is string
    && request.resource.data.snapshot.get("title", "").size() <= 120
    && request.resource.data.state == "open";
  // The operator's hand only. Two arms: the closing move (from
  // open, one way, the only moment the snapshot may be cleared)
  // and disposition corrections on a closed case (snapshot
  // excluded). When disposedBy is written, it is the writer.
  allow update: if isOwner()
    && (!("disposedBy" in request.resource.data.diff(resource.data).affectedKeys())
        || request.resource.data.disposedBy == request.auth.uid)
    && (
      (resource.data.state == "open"
        && request.resource.data.state in ["actioned", "dismissed"]
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["state", "disposition", "disposedAt", "disposedBy", "snapshot"]))
      || (resource.data.state in ["actioned", "dismissed"]
        && request.resource.data.state == resource.data.state
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["disposition", "disposedAt", "disposedBy"]))
    );
  allow delete: if false;
}
